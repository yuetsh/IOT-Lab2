export function runMockData() {
  const groups = db.query('SELECT id, name FROM groups ORDER BY name').all() as { id: number; name: string }[]
  if (groups.length === 0) throw new Error('请先加载种子数据')

  const stickers = db.query('SELECT id, name FROM stickers').all() as { id: number; name: string }[]
  const stickerByName = new Map(stickers.map(s => [s.name, s.id]))

  const BASE_TIME = Date.now() - 3 * 60 * 60 * 1000 // 3小时前开始

  for (let gi = 0; gi < groups.length; gi++) {
    const group = groups[gi]
    const groupOffset = gi * 5 * 60 * 1000 // 每组错开5分钟

    for (let ai = 0; ai < AREAS.length; ai++) {
      const area = AREAS[ai]
      const areaOffset = ai * 30 * 60 * 1000 // 每区域错开30分钟
      const baseTs = BASE_TIME + groupOffset + areaOffset

      const draftCode = AREA_DRAFT_FLOWCHARTS[area]
      const finalCode = AREA_REFERENCE_FLOWCHARTS[area]
      const checkItems = AREA_CHECK_ITEMS[area]

      // 第1轮：草稿流程图
      const h1 = db.query(
        'INSERT INTO flowchart_history (group_id, area, mermaid_code, user_prompt, created_at) VALUES (?, ?, ?, ?, ?) RETURNING id'
      ).get(group.id, area, draftCode, `请帮我生成${area}的流程图`, formatTs(new Date(baseTs))) as { id: number }

      // 第1轮检测：最后2项未通过
      const check1Results = checkItems.map((item, idx) => ({
        passed: idx < checkItems.length - 2,
        comment: idx < checkItems.length - 2
          ? '符合要求'
          : `未检测到相关节点，请补充${item.replace(/是否.*/, '')}的相关节点`,
      }))
      db.query(
        'INSERT INTO check_results (group_id, flowchart_history_id, area, results_json, created_at) VALUES (?, ?, ?, ?, ?)'
      ).run(group.id, h1.id, area, JSON.stringify(check1Results), formatTs(new Date(baseTs + 2 * 60 * 1000)))

      // 第2轮：最终流程图（参考答案）
      const h2 = db.query(
        'INSERT INTO flowchart_history (group_id, area, mermaid_code, user_prompt, created_at) VALUES (?, ?, ?, ?, ?) RETURNING id'
      ).get(group.id, area, finalCode, '已修改，请重新检查', formatTs(new Date(baseTs + 15 * 60 * 1000))) as { id: number }

      // 第2轮检测：全部通过
      const check2Results = checkItems.map(() => ({ passed: true, comment: '符合要求' }))
      db.query(
        'INSERT INTO check_results (group_id, flowchart_history_id, area, results_json, created_at) VALUES (?, ?, ?, ?, ?)'
      ).run(group.id, h2.id, area, JSON.stringify(check2Results), formatTs(new Date(baseTs + 17 * 60 * 1000)))

      // 当前流程图状态（upsert）
      const existing = db.query('SELECT group_id FROM flowcharts WHERE group_id = ? AND area = ?').get(group.id, area)
      if (existing) {
        db.query('UPDATE flowcharts SET mermaid_code = ?, updated_at = ? WHERE group_id = ? AND area = ?')
          .run(finalCode, formatTs(new Date(baseTs + 15 * 60 * 1000)), group.id, area)
      } else {
        db.query('INSERT INTO flowcharts (group_id, area, mermaid_code, updated_at) VALUES (?, ?, ?, ?)')
          .run(group.id, area, finalCode, formatTs(new Date(baseTs + 15 * 60 * 1000)))
      }

      // 聊天消息
      const msgs = AREA_MESSAGES[area]
      for (let mi = 0; mi < msgs.length; mi++) {
        db.query('INSERT INTO messages (group_id, role, content, area, created_at) VALUES (?, ?, ?, ?, ?)')
          .run(group.id, msgs[mi].role, msgs[mi].content, area, formatTs(new Date(baseTs + mi * 3 * 60 * 1000)))
      }

      // 设备提交
      const nodeMappings = AREA_DEVICE_NODE_MAPPINGS[area] ?? []
      const placements = nodeMappings.flatMap(({ node, devices }, nodeIdx) =>
        devices
          .map((deviceName, devIdx) => {
            const stickerId = stickerByName.get(deviceName) ?? null
            if (!stickerId) return null
            const stickerIdx = stickers.findIndex(s => s.name === deviceName)
            return {
              sticker_id: stickerId,
              node_id: node,
              node_label: node,
              sticker_name: deviceName,
              sticker_filename: `seed-device-${String(stickerIdx + 1).padStart(2, '0')}.svg`,
              x: 100 + nodeIdx * 120 + devIdx * 20,
              y: 100 + nodeIdx * 80,
            }
          })
          .filter(Boolean)
      )

      const sub = db.query(
        'INSERT INTO device_submissions (group_id, area, placements_json, mermaid_code, created_at) VALUES (?, ?, ?, ?, ?) RETURNING id'
      ).get(group.id, area, JSON.stringify(placements), finalCode, formatTs(new Date(baseTs + 20 * 60 * 1000))) as { id: number }

      // 设备检测结果：第6组（gi=5）第1个区域第1个设备标记为失败，其余全部通过
      const deviceCheckItems = nodeMappings.flatMap(({ node, devices }, nodeIdx) =>
        devices.map((deviceName, devIdx) => {
          const fail = gi === 5 && nodeIdx === 0 && devIdx === 0
          return {
            device_name: deviceName,
            node_label: node,
            passed: !fail,
            comment: fail
              ? `${deviceName}与节点"${node}"的对应关系有误，请重新确认`
              : '设备放置正确',
          }
        })
      )
      db.query(
        'INSERT INTO device_check_results (group_id, submission_id, area, results_json, created_at) VALUES (?, ?, ?, ?, ?)'
      ).run(group.id, sub.id, area, JSON.stringify(deviceCheckItems), formatTs(new Date(baseTs + 22 * 60 * 1000)))

      // 日志贴纸放置
      for (let ni = 0; ni < nodeMappings.length; ni++) {
        const { node, devices } = nodeMappings[ni]
        for (let di = 0; di < devices.length; di++) {
          const stickerId = stickerByName.get(devices[di])
          if (!stickerId) continue
          db.query(
            'INSERT INTO journal_placements (group_id, area, sticker_id, node_id, node_label, x, y, scale) VALUES (?, ?, ?, ?, ?, ?, ?, ?)'
          ).run(group.id, area, stickerId, node, node, 80 + ni * 130 + di * 25, 80 + ni * 90, 1.0)
        }
      }
    }
  }

  console.log('✅ 模拟数据生成完成')
  console.log(`   ${groups.length} 个小组 × ${AREAS.length} 个区域`)
}
